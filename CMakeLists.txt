cmake_minimum_required(VERSION 3.21)
project(velri VERSION 0.2.0 LANGUAGES C)

set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

include(CTest)

# The computation library lives in this repository and also builds on its own.
set(VELRI_LIB_DEFAULT_SOURCE "")
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/velri-lib/CMakeLists.txt")
    set(VELRI_LIB_DEFAULT_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/velri-lib")
endif()
set(VELRI_LIB_SOURCE_DIR "${VELRI_LIB_DEFAULT_SOURCE}" CACHE PATH "Velri-lib source directory; empty uses an installed package")
if(VELRI_LIB_SOURCE_DIR)
    get_filename_component(VELRI_LIB_SOURCE_DIR "${VELRI_LIB_SOURCE_DIR}" ABSOLUTE
        BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
    if(NOT EXISTS "${VELRI_LIB_SOURCE_DIR}/CMakeLists.txt")
        message(FATAL_ERROR "VELRI_LIB_SOURCE_DIR must point to a Velri-lib checkout")
    endif()
    set(VELRI_LIB_BUILD_TESTS ${BUILD_TESTING} CACHE BOOL "Build dependency correctness tests")
    add_subdirectory("${VELRI_LIB_SOURCE_DIR}" velri-lib)
else()
    find_package(VelriLib 0.6.0 EXACT CONFIG REQUIRED)
endif()

# Each architecture implements build_graph in a C file named after the model.
file(GLOB velri_arch_sources LIST_DIRECTORIES false CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/src/model/arch/*.c")
set(velri_generated_include_dir "${CMAKE_CURRENT_BINARY_DIR}/generated")
set(velri_arch_modules "")
set(velri_arch_declarations "")
set(velri_arch_entries "")
foreach(source IN LISTS velri_arch_sources)
    get_filename_component(arch "${source}" NAME_WLE)
    if(NOT arch MATCHES "^[A-Za-z0-9_][A-Za-z0-9_.-]*$")
        message(FATAL_ERROR "Invalid model architecture filename: ${source}")
    endif()

    # Preserve the two IDs published before filename-based discovery.
    if(arch STREQUAL "gpt2")
        set(arch_id VELRI_ARCH_GPT2)
    elseif(arch STREQUAL "llama")
        set(arch_id VELRI_ARCH_LLAMA)
    else()
        string(SHA256 digest "${arch}")
        string(SUBSTRING "${digest}" 0 8 prefix)
        math(EXPR arch_id "2 + (0x${prefix} % 2147483646)")
        if(DEFINED velri_arch_with_id_${arch_id})
            message(FATAL_ERROR "Model architectures '${arch}' and '${velri_arch_with_id_${arch_id}}' share ID ${arch_id}")
        endif()
        set(velri_arch_with_id_${arch_id} "${arch}")
    endif()

    set(module "${velri_generated_include_dir}/arch/${arch}.c")
    configure_file(src/model/arch_module.c.in "${module}" @ONLY)
    set_source_files_properties("${module}" PROPERTIES SKIP_UNITY_BUILD_INCLUSION ON)
    list(APPEND velri_arch_modules "${module}")
    string(APPEND velri_arch_declarations "extern const velri_arch_ops velri_ops_${arch_id};\n")
    string(APPEND velri_arch_entries "    &velri_ops_${arch_id},\n")
endforeach()
file(CONFIGURE OUTPUT "${velri_generated_include_dir}/models.h" CONTENT [=[
/* Generated by CMake from the model architecture sources. */
@velri_arch_declarations@
static const velri_arch_ops *const k_arch_table[] = {
@velri_arch_entries@    nullptr,
};
]=] @ONLY)

# Velri engine: public C23 runtime API, sampler, streaming JSON, cascading placement, model layers & architectures.
add_library(velri_engine STATIC
    src/format/json.c
    src/format/inspect.c
    src/config/placement.c
    src/config/placement_resolve.c
    src/config/placement_inspect.c
    src/model/layers.c
    src/model/manifest.c
    ${velri_arch_modules}
    src/model/arch_registry.c
    src/runtime/sampler.c
    src/runtime/context.c
    src/runtime/model.c
    src/runtime/weights.c
    src/runtime/program.c
    src/runtime/session.c
)
target_include_directories(velri_engine
    PUBLIC  "${CMAKE_CURRENT_SOURCE_DIR}/include"
    PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src" "${velri_generated_include_dir}"
)
target_link_libraries(velri_engine PRIVATE VelriLib::velri)
if(MSVC OR CMAKE_C_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    target_compile_options(velri_engine PRIVATE /W4 /fp:precise /std:clatest)
else()
    target_compile_options(velri_engine PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math)
endif()
if(UNIX)
    target_link_libraries(velri_engine PRIVATE m)
endif()

if(VELRI_BUILD_BENCHMARKS AND VELRI_LIB_SOURCE_DIR)
    add_executable(bench_session tools/bench_session.c)
    target_include_directories(bench_session PRIVATE tests "${VELRI_LIB_SOURCE_DIR}/src/platform")
    target_link_libraries(bench_session PRIVATE velri_engine VelriLib::velri)
endif()
